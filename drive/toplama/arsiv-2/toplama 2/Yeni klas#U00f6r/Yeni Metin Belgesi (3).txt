const CUZDAN_BASLANGIC=1000;
const ISLEM_MARJ=50;      // her pozisyon sabit 50 USDT marjin
const KALDIRAC=10;        // sabit 10x → nominal 500 USDT
const ZARAR_LIMIT=25;     // stop: −25 USDT
const KOMISYON=0.0004;    // taker %0,04 (açılış+kapanış)
const BAKIM_MARJI=0.005;

let S={sembol:'BTCUSDT',periyot:'15m',cuzdan:CUZDAN_BASLANGIC,
  pozlar:[],gecmis:[],mumlar:[],ws:null,acik:false,str:'trend',sonFiyat:null};

/* ---------- Göstergeler ---------- */
function ema(a,p){const k=2/(p+1);let e=null;const r=[];for(const v of a){e=e===null?v:v*k+e*(1-k);r.push(e)}return r}
function rsiD(a,p=14){const r=[null];let ag=null,al=null;
  for(let i=1;i<a.length;i++){const d=a[i]-a[i-1],g=Math.max(d,0),l=Math.max(-d,0);
    if(i<p){ag=(ag||0)+g;al=(al||0)+l;r.push(null)}
    else if(i===p){ag/=p;al/=p;r.push(100-100/(1+(al===0?100:ag/al)))}
    else{ag=(ag*(p-1)+g)/p;al=(al*(p-1)+l)/p;r.push(100-100/(1+(al===0?100:ag/al)))}}
  return r}
function hesapla(){
  const k=S.mumlar.map(m=>m.c);
  return{rsi:rsiD(k),e20:ema(k,20),e50:ema(k,50),e200:ema(k,200)};
}

/* ---------- Strateji sinyali (kapanmış mum) ---------- */
function sinyal(g,i){
  if(i<210)return'FLAT';
  const c=S.mumlar[i].c,r=g.rsi[i];
  const up=c>g.e20[i]&&g.e20[i]>g.e50[i]&&g.e50[i]>g.e200[i];
  const dn=c<g.e20[i]&&g.e20[i]<g.e50[i]&&g.e50[i]<g.e200[i];
  if(S.str==='trend')return up?'LONG':dn?'SHORT':'FLAT';
  if(S.str==='rsirev'){const rp=g.rsi[i-1];if(rp==null||r==null)return'FLAT';
    if(rp<35&&r>35)return'LONG';if(rp>65&&r<65)return'SHORT';return'KAPAT'}
  if(up&&r>50&&r<72)return'LONG';
  if(dn&&r<50&&r>28)return'SHORT';
  return'FLAT';
}

/* ---------- Futures pozisyon motoru ---------- */
function stopFiyati(taraf,giris,adet){
  return taraf==='LONG' ? giris-ZARAR_LIMIT/adet : giris+ZARAR_LIMIT/adet;
}
function likidasyonFiyati(taraf,giris){
  return taraf==='LONG' ? giris*(1-1/KALDIRAC+BAKIM_MARJI) : giris*(1+1/KALDIRAC-BAKIM_MARJI);
}
function pozAc(taraf,fiyat){
  if(S.cuzdan<ISLEM_MARJ){log('fl',`bakiye yetersiz — işlem için ${ISLEM_MARJ} USDT marjin gerekli (cüzdan ${F(S.cuzdan)})`);return}
  const adet=ISLEM_MARJ*KALDIRAC/fiyat;           // nominal = 50 × 10 = 500 USDT
  const p={taraf,giris:fiyat,adet,marj:ISLEM_MARJ,lev:KALDIRAC,
    stop:stopFiyati(taraf,fiyat,adet),liq:likidasyonFiyati(taraf,fiyat),saat:new Date()};
  S.pozlar.push(p);
  log('lo' , `${taraf} AÇILDI · ${fiyat.toFixed(2)} · ${adet.toFixed(5)} adet · marjin 50 USDT · 10x (nominal 500) · STOP ${p.stop.toFixed(2)} (−25 USDT)`);
  toast(`${S.sembol} ${taraf} · SL ${p.stop.toFixed(2)}`,taraf);
}
function pozKapat(p,fiyat,neden){
  const brt=p.taraf==='LONG'?(fiyat-p.giris):(p.giris-fiyat);
  const pnl=brt*p.adet-(p.giris+fiyat)*p.adet*KOMISYON;
  S.cuzdan+=pnl;
  const sira=p.taraf==='LONG'?'sh':'lo';
  const renk=pnl>=0?'lo':'sh';
  log(renk, `${p.taraf} KAPANDI · ${neden} · ${fiyat.toFixed(2)} · P&L ${pnl>=0?'+':''}${F(pnl)} USDT · ROE ${pnl>=0?'+':''}${(pnl/p.marj*100).toFixed(0)}%`);
  toast(`${neden} · P&L ${pnl>=0?'+':''}${F(pnl)}`,pnl>=0===('+',true)?(pnl>=0?'LONG':'STOP'):'STOP');
  // log için basitleştirilmiş renk:
  S.gecmis.push({taraf:p.taraf,pnl,roe:pnl/p.marj*100,neden});
}
/* Stop kontrolü — HER FİYAT GÜNCELLEMESİNDE çağrılır (mum kapanışı beklenmez) */
function stopKontrol(fiyat){
  const kalan=[];
  for(const p of S.pozlar){
    const vurdu=p.taraf==='LONG'?fiyat<=p.stop:fiyat>=p.stop;
    if(vurdu)pozKapat(p,p.stop,'STOP-LOSS');   // stop seviyesinden dolum
    else kalan.push(p);
  }
  S.pozlar=kalan;
}
/* Sinyal çıkışı — mum kapanışında değerlendirilir */
function mumKapaninca(idx){
  stopKontrol(S.mumlar[idx].c);                 // kapanışta da son kontrol
  const s=sinyal(G,idx),fiyat=S.mumlar[idx].c;
  for(const p of S.pozlar){                      // likidasyon (stop önce tetiklenir, bilgi amaçlı)
    if((p.taraf==='LONG'&&fiyat<=p.liq)||(p.taraf==='SHORT'&&fiyat>=p.liq))
      pozKapat(p,p.liq,'LİKİDASYON');
  }
  S.pozlar=S.pozlar.filter(p=>(p.taraf==='LONG'?fiyat>p.liq:fiyat<p.liq));
  if(s==='FLAT'||s==='KAPAT'){                   // SİNYAL KESİLDİ → TP noktası
    for(const p of S.pozlar)pozKapat(p,fiyat,'SİNYAL KESİLDİ');
    S.pozlar=[];return;
  }
  const kalan=[];
  for(const p of S.pozlar){
    if(p.taraf!==s)pozKapat(p,fiyat,'SİNYAL TERS DÖNDÜ'); else kalan.push(p);
  }
  S.pozlar=kalan;
  if(!S.pozlar.length)pozAc(s,fiyat);
}
function acikPnl(){
  return S.pozlar.reduce((t,p)=>{
    const brt=p.taraf==='LONG'?(S.sonFiyat-p.giris):(p.giris-S.sonFiyat);
    return t+brt*p.adet;},0);
}

/* ---------- Veri ---------- */

}

    S.sonFiyat=m.c;
    stopKontrol(S.sonFiyat);   // ★ stop: mum kapanışı beklemeden her tick'te
    guncelle();
  };
}

/* ---------- Backtest: SL −25 (fitilden) · TP sinyal sonu ---------- */
let btEs=[];
function kapatHesap(p,f){const brt=p.taraf==='LONG'?f-p.giris:p.giris-f;
  return brt*p.adet-(p.giris+f)*p.adet*KOMISYON}
function backtest(){
  let cuz=1000,pz=null,zirve=1000,dd=0,n=0,k=0,lo=0,sh=0;
  let stopC=0,sigC=0,best=-1e9,worst=1e9,top=0;
  btEs=[1000];
  for(let i=210;i<S.mumlar.length;i++){
    const m=S.mumlar[i];
    if(pz){
      // ① stop önce: mumun fitili stop'a değdi mi?
      const vurdu=pz.taraf==='LONG'?m.l<=pz.stop:m.h>=pz.stop;
      if(vurdu){
        const pnl=kapatHesap(pz,pz.stop);
        cuz+=pnl;n++;stopC++;best=Math.max(best,pnl);worst=Math.min(worst,pnl);if(pnl>0)k++;
        pz=null;
      } else {
        // ② sinyal hâlâ aynı yönde mi? Değilse → TP (sinyal kesildi)
        const s=sinyal(G,i);
        if(pz.taraf!==s){
          const pnl=kapatHesap(pz,m.c);
          cuz+=pnl;n++;sigC++;best=Math.max(best,pnl);worst=Math.min(worst,pnl);if(pnl>0)k++;
          pz=null;
        }
      }
    }
    if(!pz){
      const s=sinyal(G,i);
      if(s==='LONG'||s==='SHORT'){
        const adet=500/m.c;
        pz={taraf:s,giris:m.c,adet,stop:stopFiyati(s,m.c,adet),marj:ISLEM_MARJ};
        s==='LONG'?lo++:sh++;
      }
    }
    const an=pz?cuz+((pz.taraf==='LONG'?m.c-pz.giris:pz.giris-m.c)*pz.adet):cuz;
    zirve=Math.max(zirve,an);dd=Math.max(dd,(zirve-an)/zirve*100);
    top+=an;btEs.push(an);
  }
  if(pz){const pnl=kapatHesap(pz,S.mumlar.at(-1).c);cuz+=pnl;n++;sigC++;btEs[btEs.length-1]=cuz}
  const g=(cuz/1000-1)*100;
  $('#btG').textContent=(g>=0?'+':'')+g.toFixed(2)+'%';$('#btG').style.color=g>=0?'var(--yesil)':'var(--kirmizi)';
  $('#btI').textContent=n;
  $('#btK').textContent=n?Math.round(k/n*100)+'% ('+k+'/'+n+')':'—';
  $('#btD').textContent='-'+dd.toFixed(1)+'%';
  $('#btCk').innerHTML=`<span style="color:var(--kirmizi)">${stopC}</span> / <span style="color:var(--yesil)">${sigC}</span>`;
  $('#btOrt').textContent=n?((cuz-1000)/n).toFixed(2)+' USDT':'—';
  $('#btOzet').textContent=`${S.sembol} · ${S.periyot} · $50 · 10x · SL −$25 · ${S.mumlar.length} mum`;
  btCiz();toast('Backtest: '+(g>=0?'+':'')+g.toFixed(2)+'% · '+n+' işlem · '+stopC+' stop');
}
