/* =========================================================
   OTOMAT-F — Freqtrade futures mimarisinden uyarlanmış
   USDT-M vadeli long/short botu (paper-trading)
   Mimarisi: sinyal → pozisyon yöneticisi → marjin/PnL motoru

const $=s=>document.querySelector(s);
const F=n=>n==null?'—':Number(n).toLocaleString('tr-TR',{minimumFractionDigits:2,maximumFractionDigits:2});

const KOMISYON=0.0004, BAKIM_MARJI=0.005;   // taker %0,04 · izole marjin bakımı %0,5

let S={
  sembol:'BTCUSDT', periyot:'15m', kaldirac:10, marjOrani:1,
  cuzdan:1000, pozlar:[], gecmis:[],
  mumlar:[], ws:null, acik:false, str:'trend', sonFiyat:null
};

/* ---------- Göstergeler ---------- */
function ema(a,p){const k=2/(p+1);let e=null;const r=[];for(const v of a){e=e===null?v:v*k+e*(1-k);r.push(e)}return r}
function rsiD(a,p=14){const r=[null];let ag=null,al=null;
  for(let i=1;i<a.length;i++){const d=a[i]-a[i-1],g=Math.max(d,0),l=Math.max(-d,0);
    if(i<p){ag=(ag||0)+g;al=(al||0)+l;r.push(null)}
    else if(i===p){ag=(ag-(al===null?0:0))/1; /* первые p */ ag=ag/p;al=al/p;r.push(100-100/(1+(al===0?100:ag/al)))}
    else{ag=(ag*(p-1)+g)/p;al=(al*(p-1)+l)/p;r.push(100-100/(1+(al===0?100:ag/al)))}}
  return r}
function hesapla(){
  const k=S.mumlar.map(m=>m.c);
  return{rsi:rsiD(k),e20:ema(k,20),e50:ema(k,50),e200:ema(k,200)};
}

/* ---------- Strateji: kapanmış mumun indeksine sinyal ---------- */
function sinyal(g,i){
  if(i<210) return 'FLAT';
  const c=S.mumlar[i].c, r=g.rsi[i];
  const up=c>g.e20[i]&&g.e20[i]>g.e50[i]&&g.e50[i]>g.e200[i];
  const dn=c<g.e20[i]&&g.e20[i]<g.e50[i]&&g.e50[i]<g.e200[i];
  if(S.str==='trend')  return up?'LONG':dn?'SHORT':'FLAT';
  if(S.str==='rsirev'){
    const rp=g.rsi[i-1];
    if(rp===null||r===null)return'FLAT';
    if(rp<35&&r>35)return'LONG';
    if(rp>65&&r<65)return'SHORT';
    return'KAPAT';
  }
  // kombine: trend yönü + RSI onayı
  if(up&&r>50&&r<72)return'LONG';
  if(dn&&r<50&&r>28)return'SHORT';
  return'FLAT';
}
let G=null, cizilen=[];

function tumSinyalleriBul(){
  cizilen=[];
  for(let i=210;i<S.mumlar.length;i++){const s=sinyal(G,i);
    if(s==='LONG'||s==='SHORT')cizilen.push({i,tur:s,f:S.mumlar[i].c})}
}

/* ---------- Futures motoru ---------- */
function likidasyonFiyati(taraf,giris,lev){
  return taraf==='LONG'
    ? giris*(1-1/lev+BAKIM_MARJI)
    : giris*(1+1/lev-BAKIM_MARJI);
}
function pozAc(taraf,fiyat){
  const marj=S.cuzdan*S.marjOrani;
  if(marj<=1)return;
  const adet=marj*S.kaldirac/fiyat;
  S.pozlar.push({taraf,giris:fiyat,adet,marj,lev:S.kaldirac,
    liq:likidasyonFiyati(taraf,fiyat,S.kaldirac),saat:new Date()});
  log(taraf==='LONG'?'lo':'sh',
    `${taraf} AÇILDI · ${fiyat.toFixed(2)} · ${adet.toFixed(5)} adet · ${S.kaldirac}x · marjin ${F(marj)} USDT · liq ≈ ${S.pozlar.at(-1).liq.toFixed(2)}`);
  toast(`${S.sembol} ${taraf} · ${S.kaldirac}x kaldıraç`,taraf);
}
function pozKapat(p,fiyat){
  const brt=p.taraf==='LONG'?(fiyat-p.giris):(p.giris-fiyat);
  const ham=brt*p.adet;
  const ucret=(p.giris+fiyat)*p.adet*KOMISYON;           // açılış+kapanış taker ücreti
  const pnl=ham-ucret;
  S.cuzdan+=pnl;
  S.gecmis.push({taraf:p.taraf,pnl,roe:pnl/p.marj*100});
  log(p.taraf==='LONG'?'sh':'lo',
    `${p.taraf} KAPANDI · ${fiyat.toFixed(2)} · P&L ${pnl>=0?'+':''}${F(pnl)} USDT · ROE ${pnl>=0?'+':''}${(pnl/p.marj*100).toFixed(1)}%`);
  toast(`${p.taraf} kapandı · P&L ${pnl>=0?'+':''}${F(pnl)} USDT`,pnl>=0?'LONG':'SHORT');
}
function mumKapaninca(idx){
  if(!S.acik)return;
  const s=sinyal(G,idx), fiyat=S.mumlar[idx].c;
  /* Likidasyon kontrolü: fiyat likidasyon eşiğini geçtiyse pozisyon zorla kapanır */
  S.pozlar=S.pozlar.filter(p=>{
    const dokundu=(p.taraf==='LONG'&&fiyat<=p.liq)||(p.taraf==='SHORT'&&fiyat>=p.liq);
    if(dokundu){log('fl',`LİKİDASYON! ${p.taraf} pozisyonu ${p.liq.toFixed(2)} seviyesinde tasfiye edildi`);toast('LİKİDASYON — '+p.taraf,'SHORT')}
    return !dokundu;
  });
  if(s==='FLAT'){ S.pozlar.forEach(p=>pozKapat(p,fiyat)); S.pozlar=[]; return; }
  if(s==='KAPAT'){ S.pozlar.forEach(p=>pozKapat(p,fiyat)); S.pozlar=[]; return; }
  // ters sinyal → var olan ters pozisyonları kapat, sonra yeni aç
  const kalan=[];
  for(const p of S.pozlar){
    if(p.taraf!==s)pozKapat(p,fiyat); else kalan.push(p);
  }
  S.pozlar=kalan;
  if(!S.pozlar.length)pozAc(s,fiyat);
}
function acikPnl(){
  return S.pozlar.reduce((t,p)=>{
    const brt=p.taraf==='LONG'?(S.sonFiyat-p.giris):(p.giris-S.sonFiyat);
    return t+brt*p.adet;},0);
}

/* ---------- Veri ---------- */
async function veriCek(){
  const r=await fetch(`https://fapi.binance.com/fapi/v1/klines?symbol=${S.sembol}&interval=${S.periyot}&limit=500`);
  if(!r.ok)throw new Error('API');
  const h=await r.json();
  S.mumlar=h.map(m=>({t:m[0],o:+m[1],h:+m[2],l:+m[3],c:+m[4],v:+m[5]}));
  S.sonFiyat=S.mumlar.at(-1).c;
  G=hesapla(); tumSinyalleriBul(); $('#bd').textContent='veri yüklendi'; guncelle();
}
function ws(){
  if(S.ws)S.ws.close();
  S.ws=new WebSocket(`wss://fstream.binance.com/ws/${S.sembol.toLowerCase()}@kline_${S.periyot}`);
  S.ws.onopen=()=>{$('#led').classList.add('acik');$('#bd').textContent=`CANLI · ${S.sembol} FUTURES`};
  S.ws.onclose=()=>{$('#led').classList.remove('acik');setTimeout(ws,2000)};
  S.ws.onerror=()=>S.ws.close();
  S.ws.onmessage=e=>{
    const k=JSON.parse(e.data).k;
    const m={t:k.t,o:+k.o,h:+k.h,l:+k.l,c:+k.c,v:+k.v};
    const s=S.mumlar.at(-1);
    if(s&&k.t===s.t)S.mumlar[S.mumlar.length-1]=m;
    else if(k.t>s.t){S.mumlar.push(m);if(S.mumlar.length>600)S.mumlar.shift();
      G=hesapla();tumSinyalleriBul();mumKapaninca(S.mumlar.length-1);}
    S.sonFiyat=m.c;guncelle();
  };
}

const C=$('#anaC'),X=C.getContext('2d');let mouse=null;

/* ---------- Backtest (long+short, kaldıraçlı) ---------- */
let btEs=[];
function backtest(){
  let cuz=1000,pz=null,zirve=1000,dd=0,n=0,k=0,lo=0,sh=0,best=-1e9,worst=1e9;
  btEs=[1000];
  for(let i=210;i<S.mumlar.length;i++){
    const s=sinyal(G,i),f=S.mumlar[i].c;
    if(pz&&(pz.taraf==='LONG'?s!=='LONG':s!=='SHORT')){
      const brt=pz.taraf==='LONG'?(f-pz.g):(pz.g-f);
      const pnl=brt*pz.adet-(pz.g+f)*pz.adet*KOMISYON;
      cuz+=pnl;n++;if(pnl>0)k++;best=Math.max(best,pnl);worst=Math.min(worst,pnl);pz=null;
    }
    if(!pz&&(s==='LONG'||s==='SHORT')){
      const marj=cuz*S.marjOrani;if(marj<=1)continue;
      pz={taraf:s,giris:f,adet:marj*S.kaldirac/f};if(s==='LONG')lo++;else sh++;
    }
    const an=pz?cuz+((pz.taraf==='LONG'?(S.mumlar[i].c-pz.giris):(pz.giris-S.mumlar[i].c))*pz.adet):cuz;
    zirve=Math.max(zirve,an);dd=Math.max(dd,(zirve-an)/zirve*100);
    btEs.push(an);
  }
  const g=(cuz/1000-1)*100;
  $('#btG').textContent=(g>=0?'+':'')+g.toFixed(2)+'%';$('#btG').style.color=g>=0?'var(--yesil)':'var(--kirmizi)';
  $('#btI').textContent=n;$('#btK').textContent=n?Math.round(k/n*100)+'%':'—';
  $('#btD').textContent=n?'-'+dd.toFixed(1)+'%':'—';
  $('#btLS').textContent=`${lo} / ${sh}`;
  $('#btBest').innerHTML=`<span style="color:var(--yesil)">${best>-1e9?'+'+best.toFixed(1):'—'}</span> / <span style="color:var(--kirmizi)">${worst<1e9?worst.toFixed(1):'—'}</span>`;
  $('#btOzet').textContent=`${S.sembol} · ${S.periyot} · ${S.kaldirac}x · ${S.marjOrani*100}% marjin · ${S.mumlar.length} mum`;
  btCiz();toast('Backtest: '+(g>=0?'+':'')+g.toFixed(2)+'% · '+n+' işlem');
}
function btCiz(){
  const c=$('#btC');const W=boyut(c),x=c.getContext('2d');x.clearRect(0,0,W,210);
  if(!btEs.length)return;
  const mx=Math.max(...btEs),mn=Math.min(...btEs),r=(mx-mn)||1;
  x.strokeStyle='#141920';for(let i=0;i<4;i++){x.beginPath();x.moveTo(0,i*50+15);x.lineTo(W,i*50+15);x.stroke()}
  x.strokeStyle=btEs.at(-1)>=1000?'#0ECB81':'#F6465D';x.lineWidth=1.6;x.beginPath();

  x.stroke();
  x.strokeStyle='rgba(247,166,0,.35)';x.setLineDash([4,4]);x.beginPath();x.moveTo(0,185-((1000-mn)/r)*160);x.lineTo(W,185-((1000-mn)/r)*160);x.stroke();x.setLineDash([]);
