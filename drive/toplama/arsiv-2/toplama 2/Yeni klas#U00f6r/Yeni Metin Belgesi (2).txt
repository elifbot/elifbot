const fmtK = n => n>=1000 ? n.toLocaleString('tr-TR',{maximumFractionDigits:0}) : n.toFixed(2);

  sembol:'BTCUSDT', periyot:'15m',
  mumlar: [],           // {t,o,h,l,c,v}
  ws:null, botAcik:false,

/* ---------- Göstergeler ---------- */
function emaDizi(a,p){const k=2/(p+1);const r=[];let e=null;for(let i=0;i<a.length;i++){e=e===null?a[i]:a[i]*k+e*(1-k);r.push(e)}return r}
function rsiDizi(a,p=14){
  const r=[null];let ag=null,al=null;
  for(let i=1;i<a.length;i++){
    const d=a[i]-a[i-1],g=Math.max(d,0),l=Math.max(-d,0);
    if(i<=p){ if(i===p){ag/=p;al/=p;r.push(100-100/(1+(al===0?100:ag/al)))} else {ag=(ag||0)+g;al=(al||0)+l;r.push(null)} }
    else { ag=(ag*(p-1)+g)/p; al=(al*(p-1)+l)/p; r.push(100-100/(1+(al===0?100:ag/al))) }
  } return r;
}
function gostergeler(mumlar){
  const kapanis = mumlar.map(m=>m.c);
  const rsi = rsiDizi(kapanis);
  const m12 = emaDizi(kapanis,12), m26 = emaDizi(kapanis,26);
  const macd = kapanis.map((_,i)=>m12[i]-m26[i]);
  const sig = emaDizi(macd,9);
  const b20 = [],ust=[],alt=[];
  for(let i=0;i<kapanis.length;i++){
    if(i<19){b20.push(null);ust.push(null);alt.push(null);continue}
    let s=0;for(let j=i-19;j<=i;j++)s+=kapanis[j];
    const ort=s/20;let v=0;for(let j=i-19;j<=i;j++)v+=(kapanis[j]-ort)**2;
    const sd=Math.sqrt(v/20);
    b20.push(ort);ust.push(ort+2*sd);alt.push(ort-2*sd);
  }
  return {rsi,macd,sig,ema20:emaDizi(kapanis,20),ema50:emaDizi(kapanis,50),b20,ust,alt};
}

/* ---------- Stratejiler: mum kapanışında sinyal ---------- */
function stratejiSinyal(g,i){
  if(i<55) return null;
  const st=S.strateji, sinyaller=[];
  const rsiG=(alt,ust)=>{const p=g.rsi[i-1],n=g.rsi[i];if(p==null||n==null)return null;
    return (p<alt&&n>alt)?'AL':(p>ust&&n<ust)?'SAT':null};
  const macdG=()=>g.macd[i-1]<=g.sig[i-1]&&g.macd[i]>g.sig[i]?'AL':(g.macd[i-1]>=g.sig[i-1]&&g.macd[i]<g.sig[i])?'SAT':null;
  const bollG=()=>{const c=S.mumlar[i].c;
    return (c<g.alt[i]&&g.rsi[i]<35)?'AL':(c>g.ust[i]&&g.rsi[i]>65)?'SAT':null};
  if(st==='rsi') sinyaller.push(rsiG(30,70));
  if(st==='macd') sinyaller.push(rsiG(30,70)&&null || macdG());
  if(st==='boll') sinyaller.push(bollG());
  if(st==='oylama'){ sinyaller.push(rsiG(30,70),macdG(),bollG());
    const al=sinyaller.filter(x=>x==='AL').length, sat=sinyaller.filter(x=>x==='SAT').length;
    return al>=2?'AL':sat>=2?'SAT':null;
  }
  return sinyaller.find(x=>x)||null;
}
let sonGosterge=null, sonSinyalIdx=-1, sonCizilenSinyaller=[];

/* ---------- API ---------- */
async function mumlariCek(){
  $("#baglantiDurum").textContent='veri çekiliyor…';
  const r = await fetch(`https://api.binance.com/api/v3/klines?symbol=${S.sembol}&interval=${S.periyot}&limit=500`);
  if(!r.ok) throw new Error('API hatası');
  const h = await r.json();
  S.mumlar = h.map(m=>({t:m[0],o:+m[1],h:+m[2],l:+m[3],c:+m[4],v:+m[5]}));
  S.sonFiyat = S.mumlar.at(-1).c;
  sonGosterge = gostergeler(S.mumlar);
  sinyalleriHesapla();
  $("#baglantiDurum").textContent='veri yüklendi';
  guncelle();
}
function sinyalleriHesapla(){
  sonCizilenSinyaller=[];
  const g=sonGosterge;
  for(let i=55;i<S.mumlar.length;i++){
    const s=stratejiSinyal(g,i);
    if(s) sonCizilenSinyaller.push({i,tur:s,fiyat:S.mumlar[i].c});
  }
}
function wsBaglan(){
  if(S.ws) S.ws.close();
  const url=`wss://stream.binance.com:9443/ws/${S.sembol.toLowerCase()}@kline_${S.periyot}`;
  S.ws=new WebSocket(url);
  S.ws.onopen=()=>{$("#led").classList.add('acik');$("#baglantiDurum").textContent=`canlı · ${S.sembol}`};
  S.ws.onclose=()=>{$("#led").classList.remove('acik');setTimeout(wsBaglan,2000)};
  S.ws.onerror=()=>S.ws.close();
  S.ws.onmessage=e=>{
    const k=JSON.parse(e.data).k;
    const mum={t:k.t,o:+k.o,h:+k.h,l:+k.l,c:+k.c,v:+k.v};
    const son=S.mumlar.at(-1);
    if(son && k.t===son.t){ S.mumlar[S.mumlar.length-1]=mum }
    else if(k.t>son.t){ 
      // önceki mum KAPANDI → bot karar anı
      S.mumlar.push(mum);
      if(S.mumlar.length>600)S.mumlar.shift();
      sonGosterge=gostergeler(S.mumlar);
      sinyalleriHesapla();
      mumKapandi(S.mumlar.length-1);
    }
    S.sonFiyat=mum.c;
    guncelle();
  };
}

/* ---------- İşlem motoru (paper trading) ---------- */
function mumKapandi(idx){
  if(!S.botAcik) return;
  const sinyal = stratejiSinyal(sonGosterge, idx);
  if(!sinyal) return;
  const fiyat = S.mumlar[idx].c;
  if(sinyal==='AL' && !S.pozisyon){
    const miktar = S.bakiye*0.98/fiyat;
    S.pozisyon={giris:fiyat,miktar,zaman:new Date()};
    log('AL',`ALIM · $${fmt(fiyat)} · ${miktar.toFixed(6)} ${S.sembol.replace('USDT','')} · bakiyenin %98'i`);
    toast(`${S.sembol} ALIM @ $${fmt(fiyat)}`,'toplu');
  } else if(sinyal==='SAT' && S.pozisyon){
    const cikis=fiyat*(1-0.001); // %0.1 komisyon
    const pnl=(cikis-S.pozisyon.giris)*S.pozisyon.miktar;
    S.bakiye+=S.pozisyon.miktar*cikis;
    S.tamamlanan.push({pnl});
    log('SAT',`SATIM · $${fmt(fiyat)} · P&L ${pnl>=0?'+':''}$${fmt(pnl)}`);
    toast(`${S.sembol} SATIM @ $${fmt(fiyat)} — P&L ${pnl>=0?'+':''}$${fmt(pnl)}`,'satti');
    S.pozisyon=null;
  }
}
function acilabilirPnl(){
  return S.pozisyon ? (S.sonFiyat-S.pozisyon.giris)*S.pozisyon.miktar : 0;
}

/* ---------- Backtest ---------- */
function backtest(){
  const g=sonGosterge, sonuc={es:[10000]}, islemler=[]; let bakiye=10000;
  let zirve=10000, maxDusus=0;
  for(let i=55;i<S.mumlar.length;i++){
    // kaba güncelleme: sinyal yoksa pozisyonu taşı
    const s=satriskBak(g,i);
    if(s==='AL'&&!sonuc.poz){sonuc.poz=S.mumlar[i].c}
    if(s==='SAT'&&sonuc.poz){
      const c=S.mumlar[i].c*(1-0.001);
      bakiye*= c/sonuc.poz; islemler.push(c>sonuc.poz);
      sonuc.poz=null;
    }
    const anlik = sonuc.poz? bakiye*(S.mumlar[i].c/sonuc.poz) : bakiye;
    zirve=Math.max(zirve,anlik);
    maxDusus=Math.max(maxDusus,(zirve-anlik)/zirve*100);
    sonuc.es.push(anlik);
  }
  const getiri=(bakiye/10000-1)*100;
  $('#btGetiri').textContent=(getiri>=0?'+':'')+getiri.toFixed(2)+'%';
  $('#btGetiri').style.color=getiri>=0?'var(--yesil)':'var(--kirmizi)';
  $('#btIslem').textContent=islemler.length;
  const k=islemler.filter(Boolean).length;
  $('#btKazanc')=islemler.length?`${(k/islemler.length*100).toFixed(0)}% (${k}/${islemler.length})`:'—';
  $('#btDusus').textContent='-'+maxDusus.toFixed(2)+'%';
  $('#btOzet').textContent=`${S.sembol} · ${S.periyot} · son ${S.mumlar.length} mum`;
  btEs=sonuc.es; btCiz();
  toast('Backtest tamamlandı: '+(getiri>=0?'+':'')+getiri.toFixed(2)+'% getiri');
}
