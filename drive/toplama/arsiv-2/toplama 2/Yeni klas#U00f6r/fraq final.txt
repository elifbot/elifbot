function acikPnlToplam(){

  for(const s of syms.values()){
    if(!s.pos||s.last==null)continue;
    t+=((s.pos.taraf==='LONG'?(s.last-s.pos.giris):(s.pos.giris-s.last)))*s.pos.adet-KOM(s.pos,s.last);
  }
  return t;
}

/* ---------- Mum akışı ---------- */
function mumGuncelle(s,m){
  const son=s.c[s.c.length-1];
  if(!son||m.t>son.t){
    s.c.push(m);
    if(!m.final){s.last=m.c;tickSL(s,m.c)}else kapanis(s,m);
  }else if(m.t===son.t){
    s.c[s.c.length-1]=m;
    if(!m.final){s.last=m.c;tickSL(s,m.c)}else kapanis(s,m);
  }
  if(s.c.length>320)s.c.splice(0,s.c.length-320);
}
function kapanis(s,m){
  /* #7 — aynı mumun (ör. OKX'in geç gönderdiği son kapanan mumun)
     ikinci kez işlenmesini engeller; updState artık çift sayamaz */
  if(s._sonKapanisT===m.t)return;
  s._sonKapanisT=m.t;
  s.last=m.c;tickSL(s,m.c);
  updState(s.st,m.c);
  CN.mum++;
  const sig=sinyalHesapla(s);s.sig=sig;
  if(sig==='LONG'||sig==='SHORT'){
    CN.sinyal++;
    feed.unshift({zaman:Date.now(),side:sig,px:m.c,s});
    if(feed.length>60)feed.pop();
  }
  if(!botOn||!hazirMi(s.st))return;
  if(s.pos){
    if(sig!==s.pos.taraf)pozKapat(s,m.c,(sig==='FLAT'||sig==='KAPAT')?'SİNYAL KESİLDİ':'SİNYAL TERS DÖNDÜ');
  }else if((sig==='LONG'||sig==='SHORT')&&acikPozSayisi()<MAXPOS&&WALLET.serbest>=CFG.MARJ){
    pozAc(s,sig,m.c);
  }
}

/* ---------- Görev havuzu — borsa bazlı hız limiti + retry ---------- */
async function havuz(isler,es,aralik,tamamlanan){
  let i=0;
  const calis=async()=>{
    while(i<isler.length){
      const j=i++;let ok=false;
      for(let den=0;den<3&&!ok;den++){
        try{await isler[j]();ok=true}catch(e){if(den<2)await sleep(600+den*600)}
      }
      if(ok)tamamlanan.ok++;else tamamlanan.hata++;
      if(aralik)await sleep(aralik);
    }
  };
  await Promise.all(Array.from({length:es},calis));
}

/* ---------- Geçmiş yükleme — çalışan ilerleme çubuğu (#2b) ---------- */
async function gecmisYukle(liste,ilerleme){
  const h=tfHarita();                       /* #1: tek satır, tfHarita */
  const islerB=[],islerO=[];
  for(const key of liste){
    const [ex,id]=key.split(':');
    const s={ex,id,key,
      disp:ex==='B'?id.replace('USDT','/USDT'):id.replace('-SWAP','').replace('-','/')+'/USDT',
      exEtiket:ex==='B'?'BİNANCE':'OKX',
      c:[],last:null,st:yeniState(),pos:null,sig:'FLAT',chg:null,_ok:false,_sonKapanisT:null};
    syms.set(key,s);
    if(ex==='B')islerB.push(async()=>{
      const r=await fetch(`https://fapi.binance.com/fapi/v1/klines?symbol=${id}&interval=${h.b}&limit=300`);
      if(!r.ok)throw new Error('http');
      const d=await r.json();
      if(!Array.isArray(d)||d.length<220)throw new Error('az veri');
      const simdi=Date.now();
      s.c=d.map(m=>({t:m[0],o:+m[1],h:+m[2],l:+m[3],c:+m[4],v:+m[5],
        final:+m[6]<simdi}));                /* closeTime<now: kesin kapanış bayrağı */
      for(const m of s.c)if(m.final)updState(s.st,m.c);
      s.last=s.c[s.c.length-1].c;
      s._sonKapanisT=s.c.filter(m=>m.final).length?s.c.filter(m=>m.final).slice(-1)[0].t:null;
      s.sig=sinyalHesapla(s);s._ok=true;
    });
    else islerO.push(async()=>{
      const r=await fetch(`https://www.okx.com/api/v5/market/candles?instId=${id}&bar=${h.o}&limit=300`);
      const j=await r.json();
      if(!j.data||j.data.length<220)throw new Error('az veri');
      s.c=j.data.map(x=>({t:+x[0],o:+x[1],h:+x[2],l:+x[3],c:+x[4],v:+x[5],
        final:x[8]==='1'})).reverse();       /* confirm bayrağı */
      for(const m of s.c)if(m.final)updState(s.st,m.c);
      s.last=s.c[s.c.length-1].c;
      s._sonKapanisT=s.c.filter(m=>m.final).length?s.c.filter(m=>m.final).slice(-1)[0].t:null;
      s.sig=sinyalHesapla(s);s._ok=true;
    });
  }
  const dB={ok:0,hata:0},dO={ok:0,hata:0};
  const rapor=()=>ilerleme(dB.ok+dO.hata+dO.ok+dB.hata, islerB.length+islerO.length, dB.hata+dO.hata);
  await Promise.all([
    havuz(islerB,12,0,dB).then(rapor),      /* Binance: 12 paralel */
    havuz(islerO,4,120,dO).then(rapor)       /* OKX: 4 paralel + 120ms → ~14 istek/2sn */
  ]);
  for(const [k,s] of [...syms]){if(!s._ok)syms.delete(k)}
}

    setTimeout(()=>{if(myGen===gen)bwsGrup(parca,gen,false)},2000);   /* reconnect: bTop şişmez */
